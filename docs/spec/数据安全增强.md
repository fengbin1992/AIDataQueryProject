# 数据安全增强 - 设计文档

> 实现敏感数据脱敏显示和访问权限审批功能

## 1. 功能概述

### 1.1 核心需求

| 角色 | 数据访问权限 | 说明 |
|------|------------|------|
| 管理员 | 完整数据 | 可查看所有字段的原始数据 |
| 普通用户（已授权） | 完整数据 | 对已审批的敏感字段可查看原始数据 |
| 普通用户（未授权） | 脱敏数据 | 敏感字段自动脱敏显示 |

### 1.2 功能特性

- **脱敏规则配置**：支持字段名模式匹配 + 手动标记敏感字段
- **审批粒度**：按敏感字段/表级别申请权限
- **权限时效**：审批通过后永久有效（可由管理员撤销）
- **审批流程**：单级审批（管理员直接审批）

## 2. 核心概念

### 2.1 敏感字段识别

```
┌─────────────────────────────────────────────────────────────┐
│                     敏感字段识别                              │
├─────────────────────────────────────────────────────────────┤
│                                                             │
│   方式一：模式匹配规则（全局生效）                             │
│   ┌─────────────────────────────────────────────────────┐   │
│   │ 规则名称      匹配模式                 脱敏类型      │   │
│   │ ─────────────────────────────────────────────────── │   │
│   │ 手机号        *phone*, *mobile*        中间4位      │   │
│   │ 身份证        *idcard*, *id_no*        中间10位     │   │
│   │ 邮箱          *email*, *mail*          用户名部分   │   │
│   │ 银行卡        *bank*, *card_no*        中间8位      │   │
│   └─────────────────────────────────────────────────────┘   │
│                                                             │
│   方式二：手动标记（针对特定连接/表）                          │
│   ┌─────────────────────────────────────────────────────┐   │
│   │ 连接          表名           字段名       脱敏类型   │   │
│   │ ─────────────────────────────────────────────────── │   │
│   │ 生产库        users          real_name    姓名脱敏  │   │
│   │ 生产库        orders         address      地址脱敏  │   │
│   │ 业务库        customers      salary       金额脱敏  │   │
│   └─────────────────────────────────────────────────────┘   │
│                                                             │
└─────────────────────────────────────────────────────────────┘
```

### 2.2 脱敏类型

| 类型 | 原始值 | 脱敏后 | 说明 |
|------|--------|--------|------|
| 手机号 | 13812345678 | 138****5678 | 保留前3后4 |
| 身份证 | 310123199001011234 | 310***********1234 | 保留前3后4 |
| 邮箱 | zhangsan@example.com | z***@example.com | 用户名保留首字母 |
| 银行卡 | 6222021234567890123 | 6222****0123 | 保留前4后4 |
| 姓名 | 张三丰 | 张** | 保留姓氏 |
| 地址 | 北京市朝阳区xxx路123号 | 北京市朝阳区****** | 保留省市区 |
| 金额 | 50000.00 | ****** | 完全隐藏 |
| 自定义 | 任意 | 按配置规则 | 正则替换 |

```
┌──────────────┐     ┌──────────────┐     ┌──────────────┐
│   普通用户    │────▶│  提交申请    │────▶│  等待审批    │
└──────────────┘     └──────────────┘     └──────────────┘
                                                 │

┌──────────────┐     ┌──────────────┐     ┌──────────────┐
│   查看原数据  │◀────│  审批通过    │◀────│  管理员审批  │
└──────────────┘     └──────────────┘     └──────────────┘
        │                                        │
        │                                        ▼

└──────────────┘                       └──────────────┘
```

## 3. 数据库设计

### 3.1 新增表结构

```sql
-- =====================================================
-- 脱敏规则表（全局模式匹配规则）
-- =====================================================
CREATE TABLE SensitiveMaskingRules (
    Id INT PRIMARY KEY AUTO_INCREMENT,
    Name NVARCHAR(100) NOT NULL COMMENT '规则名称',
    FieldPattern NVARCHAR(200) NOT NULL COMMENT '字段名匹配模式，支持通配符，多个用逗号分隔',
    MaskType NVARCHAR(50) NOT NULL COMMENT '脱敏类型：phone/idcard/email/bankcard/name/address/amount/custom',
    MaskConfig JSON NULL COMMENT '自定义脱敏配置，如正则表达式',
    Priority INT DEFAULT 0 COMMENT '优先级，数字越大优先级越高',
    IsActive BIT DEFAULT 1 COMMENT '是否启用',
    Description NVARCHAR(500) NULL COMMENT '规则说明',
    CreatedAt DATETIME DEFAULT CURRENT_TIMESTAMP,
    UpdatedAt DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    INDEX idx_active_priority (IsActive, Priority DESC)
);

-- =====================================================
-- 敏感字段标记表（手动标记特定字段）
-- =====================================================
CREATE TABLE SensitiveFieldMarks (
    Id INT PRIMARY KEY AUTO_INCREMENT,
    ConnectionId INT NOT NULL COMMENT '数据库连接ID',
    TableName NVARCHAR(200) NOT NULL COMMENT '表名，支持通配符 * 表示所有表',
    FieldName NVARCHAR(200) NOT NULL COMMENT '字段名',
    MaskType NVARCHAR(50) NOT NULL COMMENT '脱敏类型',
    MaskConfig JSON NULL COMMENT '自定义脱敏配置',
    Description NVARCHAR(500) NULL COMMENT '标记说明',
    MarkedBy INT NOT NULL COMMENT '标记人ID',
    CreatedAt DATETIME DEFAULT CURRENT_TIMESTAMP,
    UpdatedAt DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    FOREIGN KEY (ConnectionId) REFERENCES DatabaseConnections(Id) ON DELETE CASCADE,
    FOREIGN KEY (MarkedBy) REFERENCES Users(Id),
    UNIQUE KEY uk_conn_table_field (ConnectionId, TableName, FieldName)
);

-- =====================================================
-- 敏感数据访问申请表
-- =====================================================
CREATE TABLE SensitiveAccessRequests (
    Id INT PRIMARY KEY AUTO_INCREMENT,
    UserId INT NOT NULL COMMENT '申请人ID',
    ConnectionId INT NOT NULL COMMENT '数据库连接ID',
    TableName NVARCHAR(200) NULL COMMENT '申请的表名，NULL表示申请连接下所有敏感字段',
    FieldNames JSON NULL COMMENT '申请的字段列表，NULL表示申请表下所有敏感字段',
    Reason NVARCHAR(1000) NOT NULL COMMENT '申请原因',
    Status TINYINT DEFAULT 0 COMMENT '状态：0-待审批，1-已通过，2-已拒绝，3-已撤销',
    ReviewerId INT NULL COMMENT '审批人ID',
    ReviewComment NVARCHAR(500) NULL COMMENT '审批意见',
    ReviewedAt DATETIME NULL COMMENT '审批时间',
    CreatedAt DATETIME DEFAULT CURRENT_TIMESTAMP,
    UpdatedAt DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    FOREIGN KEY (UserId) REFERENCES Users(Id),
    FOREIGN KEY (ConnectionId) REFERENCES DatabaseConnections(Id) ON DELETE CASCADE,
    FOREIGN KEY (ReviewerId) REFERENCES Users(Id),
    INDEX idx_user_status (UserId, Status),
    INDEX idx_status_created (Status, CreatedAt DESC)
);

-- =====================================================
-- 敏感数据访问权限表（审批通过后生成）
-- =====================================================
CREATE TABLE SensitiveAccessPermissions (
    Id INT PRIMARY KEY AUTO_INCREMENT,
    UserId INT NOT NULL COMMENT '用户ID',
    ConnectionId INT NOT NULL COMMENT '数据库连接ID',
    TableName NVARCHAR(200) NULL COMMENT '表名，NULL表示连接下所有表',
    FieldName NVARCHAR(200) NULL COMMENT '字段名，NULL表示表下所有敏感字段',
    RequestId INT NULL COMMENT '关联的申请ID',
    GrantedBy INT NOT NULL COMMENT '授权人ID',
    GrantedAt DATETIME DEFAULT CURRENT_TIMESTAMP,
    RevokedAt DATETIME NULL COMMENT '撤销时间',
    RevokedBy INT NULL COMMENT '撤销人ID',
    RevokeReason NVARCHAR(500) NULL COMMENT '撤销原因',

    FOREIGN KEY (UserId) REFERENCES Users(Id),
    FOREIGN KEY (ConnectionId) REFERENCES DatabaseConnections(Id) ON DELETE CASCADE,
    FOREIGN KEY (RequestId) REFERENCES SensitiveAccessRequests(Id),
    FOREIGN KEY (GrantedBy) REFERENCES Users(Id),
    FOREIGN KEY (RevokedBy) REFERENCES Users(Id),
    INDEX idx_user_conn (UserId, ConnectionId),
    INDEX idx_user_active (UserId, RevokedAt)
);

-- =====================================================
-- 敏感数据访问日志表
-- =====================================================
CREATE TABLE SensitiveAccessLogs (
    Id BIGINT PRIMARY KEY AUTO_INCREMENT,
    UserId INT NOT NULL COMMENT '访问用户ID',
    ConnectionId INT NOT NULL COMMENT '数据库连接ID',
    QueryLogId BIGINT NULL COMMENT '关联的查询日志ID',
    AccessedFields JSON NOT NULL COMMENT '访问的敏感字段列表',
    WasMasked BIT NOT NULL COMMENT '是否脱敏显示',
    AccessedAt DATETIME DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (UserId) REFERENCES Users(Id),
    FOREIGN KEY (ConnectionId) REFERENCES DatabaseConnections(Id) ON DELETE CASCADE,
    INDEX idx_user_time (UserId, AccessedAt DESC),
    INDEX idx_conn_time (ConnectionId, AccessedAt DESC)
);
```

## 4. API 设计

### 4.1 脱敏规则管理

```
# 获取脱敏规则列表
GET /api/admin/masking-rules
Response: {
  items: [
    {
      id: 1,
      name: "手机号码",
      fieldPattern: "*phone*,*mobile*",
      maskType: "phone",
      priority: 100,
      isActive: true,
      description: "..."
    }
  ]
}

# 创建脱敏规则
POST /api/admin/masking-rules
Body: {
  name: "自定义规则",
  fieldPattern: "*custom_field*",
  maskType: "custom",
  maskConfig: { pattern: "(.{3}).*(.{4})", replacement: "$1****$2" },
  priority: 50,
  description: "..."
}

# 更新脱敏规则
PUT /api/admin/masking-rules/{id}

# 删除脱敏规则
DELETE /api/admin/masking-rules/{id}

# 切换规则状态
PUT /api/admin/masking-rules/{id}/toggle
```

### 4.2 敏感字段标记

```
# 获取连接下的敏感字段标记
GET /api/admin/sensitive-fields?connectionId={id}
Response: {
  items: [
    {
      id: 1,
      connectionId: 1,
      tableName: "users",
      fieldName: "real_name",
      maskType: "name",
      description: "用户真实姓名"
    }
  ]
}

# 批量标记敏感字段
POST /api/admin/sensitive-fields/batch
Body: {
  connectionId: 1,
  fields: [
    { tableName: "users", fieldName: "phone", maskType: "phone" },
    { tableName: "users", fieldName: "email", maskType: "email" }
  ]
}

# 删除敏感字段标记
DELETE /api/admin/sensitive-fields/{id}

# 获取表结构并标识敏感字段（辅助标记）
GET /api/admin/sensitive-fields/schema?connectionId={id}&tableName={name}
Response: {
  tableName: "users",
  fields: [
    { name: "id", type: "int", isSensitive: false, matchedRule: null },
    { name: "phone", type: "varchar", isSensitive: true, matchedRule: "手机号码" },
    { name: "custom_field", type: "varchar", isSensitive: false, matchedRule: null }
  ]
}
```

### 4.3 访问权限申请

```
# 获取我的申请列表
GET /api/sensitive-access/my-requests?status={status}
Response: {
  items: [
    {
      id: 1,
      connectionName: "生产数据库",
      tableName: "users",
      fieldNames: ["phone", "email"],
      reason: "需要核对用户联系方式",
      status: 0,
      statusText: "待审批",
      createdAt: "2026-02-02T10:00:00"
    }
  ]
}

# 提交访问申请
POST /api/sensitive-access/requests
Body: {
  connectionId: 1,
  tableName: "users",        // 可选，不填表示申请连接下所有
  fieldNames: ["phone"],     // 可选，不填表示申请表下所有敏感字段
  reason: "申请原因说明"
}

# 撤销我的申请（仅待审批状态可撤销）
DELETE /api/sensitive-access/requests/{id}

# 获取我的已授权权限
GET /api/sensitive-access/my-permissions
Response: {
  items: [
    {
      id: 1,
      connectionId: 1,
      connectionName: "生产数据库",
      tableName: "users",
      fieldName: "phone",
      grantedAt: "2026-02-01T10:00:00",
      grantedBy: "管理员"
    }
  ]
}
```

### 4.4 审批管理（管理员）

```
# 获取待审批列表
GET /api/admin/sensitive-access/pending
Response: {
  items: [
    {
      id: 1,
      userId: 2,
      userName: "张三",
      connectionName: "生产数据库",
      tableName: "users",
      fieldNames: ["phone", "email"],
      reason: "需要核对用户联系方式",
      createdAt: "2026-02-02T10:00:00"
    }
  ]
}

# 审批通过
POST /api/admin/sensitive-access/requests/{id}/approve
Body: {
  comment: "审批意见（可选）"
}

# 审批拒绝
POST /api/admin/sensitive-access/requests/{id}/reject
Body: {
  comment: "拒绝原因"
}

# 获取用户权限列表
GET /api/admin/sensitive-access/permissions?userId={id}

# 撤销用户权限
DELETE /api/admin/sensitive-access/permissions/{id}
Body: {
  reason: "撤销原因"
}

# 直接授权（无需申请）
POST /api/admin/sensitive-access/grant
Body: {
  userId: 2,
  connectionId: 1,
  tableName: "users",    // 可选
  fieldName: "phone"     // 可选
}

# 获取敏感数据访问日志
GET /api/admin/sensitive-access/logs?userId={id}&connectionId={id}&startDate={date}&endDate={date}
```

```csharp
// Services/Security/DataMaskingService.cs

public interface IDataMaskingService
{
    /// <summary>
    /// 对查询结果进行脱敏处理
    /// </summary>
    Task<MaskedQueryResult> MaskQueryResultAsync(
        QueryResult result,
        int userId,
        int connectionId);

    /// <summary>
    /// 检查字段是否需要脱敏
    /// </summary>
    Task<List<SensitiveFieldInfo>> GetSensitiveFieldsAsync(
        int connectionId,
        List<string> fieldNames);
}

public class DataMaskingService : IDataMaskingService
{
    public async Task<MaskedQueryResult> MaskQueryResultAsync(
        QueryResult result,
        int userId,
        int connectionId)
    {
        // 1. 获取用户权限
        var permissions = await GetUserPermissionsAsync(userId, connectionId);

        // 2. 识别结果中的敏感字段
        var sensitiveFields = await GetSensitiveFieldsAsync(
            connectionId,
            result.Columns.ToList());

        // 3. 确定需要脱敏的字段（敏感字段 - 已授权字段）
        var fieldsToMask = sensitiveFields
            .Where(f => !HasPermission(permissions, f))
            .ToList();

        // 4. 执行脱敏
        var maskedRows = result.Rows.Select(row =>
            MaskRow(row, fieldsToMask)).ToList();

        return new MaskedQueryResult
        {
            Columns = result.Columns,
            Rows = maskedRows,
            MaskedFields = fieldsToMask.Select(f => f.FieldName).ToList(),
            TotalRows = result.TotalRows
        };
    }

    private string MaskValue(string value, string maskType)
    {
        if (string.IsNullOrEmpty(value)) return value;

        return maskType switch
        {
            "phone" => MaskPhone(value),
            "idcard" => MaskIdCard(value),
            "email" => MaskEmail(value),
            "bankcard" => MaskBankCard(value),
            "name" => MaskName(value),
            "address" => MaskAddress(value),
            "amount" => "******",
            "full" => "******",
            _ => MaskDefault(value)
        };
    }

    private string MaskPhone(string value)
    {
        // 138****5678
        if (value.Length >= 11)
            return value.Substring(0, 3) + "****" + value.Substring(7);
        return MaskDefault(value);
    }

    private string MaskIdCard(string value)
    {
        // 310***********1234
        if (value.Length >= 15)
            return value.Substring(0, 3) + new string('*', value.Length - 7) + value.Substring(value.Length - 4);
        return MaskDefault(value);
    }

    private string MaskEmail(string value)
    {
        // z***@example.com
        var atIndex = value.IndexOf('@');
        if (atIndex > 0)
            return value[0] + "***" + value.Substring(atIndex);
        return MaskDefault(value);
    }

    // ... 其他脱敏方法
}
```

### 6.2 敏感字段识别

```csharp
public async Task<List<SensitiveFieldInfo>> GetSensitiveFieldsAsync(
    int connectionId,
    List<string> fieldNames)
{
    var result = new List<SensitiveFieldInfo>();

    // 1. 获取全局脱敏规则
    var rules = await _context.SensitiveMaskingRules
        .Where(r => r.IsActive)
        .OrderByDescending(r => r.Priority)
        .ToListAsync();

    // 2. 获取手动标记的字段
    var marks = await _context.SensitiveFieldMarks
        .Where(m => m.ConnectionId == connectionId)
        .ToListAsync();

    foreach (var fieldName in fieldNames)
    {
        // 优先检查手动标记
        var mark = marks.FirstOrDefault(m =>
            MatchFieldName(m.FieldName, fieldName) &&
            (m.TableName == "*" || MatchTableName(m.TableName)));

        if (mark != null)
        {
            result.Add(new SensitiveFieldInfo
            {
                FieldName = fieldName,
                MaskType = mark.MaskType,
                Source = "manual"
            });
            continue;
        }

        // 检查模式匹配规则
        foreach (var rule in rules)
        {
            var patterns = rule.FieldPattern.Split(',');
            if (patterns.Any(p => MatchPattern(fieldName, p.Trim())))
            {
                result.Add(new SensitiveFieldInfo
                {
                    FieldName = fieldName,
                    MaskType = rule.MaskType,
                    Source = "rule",
                    RuleName = rule.Name
                });
                break;
            }
        }
    }

    return result;
}

private bool MatchPattern(string fieldName, string pattern)
{
    // 支持通配符匹配：*phone* 匹配 user_phone, phone_number 等
    var regex = "^" + Regex.Escape(pattern)
        .Replace("\\*", ".*")
        .Replace("\\?", ".") + "$";
    return Regex.IsMatch(fieldName, regex, RegexOptions.IgnoreCase);
}
```

### 6.3 权限检查中间件

```csharp
// 在查询执行服务中集成脱敏处理
public class QueryExecutionService
{
    private readonly IDataMaskingService _maskingService;

    public async Task<QueryResult> ExecuteQueryAsync(
        int userId,
        int connectionId,
        string sql)
    {
        // 1. 执行查询
        var rawResult = await ExecuteRawQueryAsync(connectionId, sql);

        // 2. 检查是否需要脱敏（管理员跳过）
        if (await IsAdminAsync(userId))
        {
            return rawResult;
        }

        // 3. 执行脱敏
        var maskedResult = await _maskingService.MaskQueryResultAsync(
            rawResult,
            userId,
            connectionId);

        // 4. 记录访问日志
        if (maskedResult.MaskedFields.Any())
        {
            await LogSensitiveAccessAsync(userId, connectionId, maskedResult);
        }

        return maskedResult;
    }
}
```
